/*
 * SNIFF - TinyML gas detector with Wi-Fi dashboard
 * Author: Ayaan Khan
 *
 * Implements the full output stage of Figure 1 in the project proposal:
 *   Sensing    MQ-6 + DHT22 -> ADC
 *   Processing ESP32 data acquisition and feature extraction
 *   Inference  TinyML model (TensorFlow Lite Micro), safe vs gas leak
 *   Output     Alarm (LED + buzzer), Safe state (green LED), IoT alert (Wi-Fi)
 *
 * The board raises its own Wi-Fi access point, so the dashboard works with
 * no router, no internet and no credentials. Connect a phone or laptop to
 *     SSID  SNIFF-Detector
 *     pass  sniffgas2026
 * and open  http://192.168.4.1
 *
 * The page shows the model's verdict beside the conventional fixed-threshold
 * verdict, which is the comparison the project exists to make.
 *
 * Pin map (as wired):
 *   GPIO34  MQ-6 AOUT through a 10k/10k divider   (ADC1; ADC2 dies with Wi-Fi)
 *   GPIO27  DHT22 data
 *   GPIO26  DHT22 power, held HIGH
 *   GPIO33  red LED   + 220R
 *   GPIO25  green LED + 220R
 *   GPIO13  active buzzer
 *
 * Library needed (Arduino Library Manager): "TensorFlowLite_ESP32"
 */

// ------------------------------------------------------------ pins
#define MQ_PIN      34
#define DHT_PIN     27
#define DHT_PWR     26
#define LED_ALARM   33
#define LED_SAFE    25
#define BUZZER_PIN  13

// ------------------------------------------------------------ constants
const char *AP_SSID = "SNIFF-Detector";
const char *AP_PASS = "sniffgas2026";

const float VC = 5.0f;
const float RL_KOHM = 10.0f;        // MEASURE YOURS
const float DIVIDER_RATIO = 2.0f;

const int   ADC_SAMPLES = 32;
const int   SLOPE_WINDOW = 10;
const unsigned long LOOP_MS = 1000;
const unsigned long DHT_MIN_MS = 2000;

// The conventional detectors' constants are GENERATED by train_tflite.py into
// model_data.h, refitted on the full dataset. They are not hand-set here, so
// the firmware cannot drift out of sync with the analysis.
//
//   Detector 1, fixed threshold : alarm when rs_ro < SNIFF_FIXED_T
//   Detector 2, compensated     : expected = A + B*T + C*RH
//                                 alarm when (rs_ro - expected) < SNIFF_COMP_T
//
// Detector 2 is the one that beat both the fixed threshold and the network on
// held-out sessions, so the dashboard shows all three side by side.
#define BASELINE_THRESHOLD SNIFF_FIXED_T
const int   CONFIRM_SAMPLES = 3;          // debounce before latching the alarm

// Power-on warm-up lockout.
// The MQ-6's sensing element must reach operating temperature before Rs means
// anything. Cold, its resistance sits far below Ro, so rs_ro looks like a
// large gas concentration and every detector calls a leak. Commercial units
// suppress the alarm through this window for the same reason. Readings are
// still computed, served and printed during warm-up; only the outputs wait.
const unsigned long WARMUP_MS = 90000;
const int   HIST = 120;                   // seconds of history for the chart

// ------------------------------------------------------------ globals
DHT dht(DHT_PIN, DHT22);
Preferences prefs;
WebServer server(80);

float Ro_kohm = 0.0f;
float rsWindow[SLOPE_WINDOW];
int   wIdx = 0;
bool  wFull = false;

float lastTemp = NAN, lastRH = NAN;
unsigned long lastDht = 0, lastLoop = 0;
int  leakStreak = 0;
bool muted = false;

float  curRsRo = NAN, curSlope = 0, curConf = 0;
bool   curAlarm = false, curThresh = false, curComp = false;
unsigned long testUntil = 0;   // alarm self-test expiry
float  curResid = 0.0f;   // rs_ro minus its temperature/humidity expectation
uint32_t sampleCount = 0;

float  hist[HIST];
int    histN = 0, histHead = 0;

// ------------------------------------------------------------ TFLM
namespace {
  const tflite::Model *model = nullptr;
  tflite::MicroInterpreter *interpreter = nullptr;
  TfLiteTensor *inTensor = nullptr;
  TfLiteTensor *outTensor = nullptr;
  constexpr int kArenaSize = 16 * 1024;
  alignas(16) uint8_t arena[kArenaSize];
}

static bool tfliteReady = false;

static void setupModel() {
  model = tflite::GetModel(sniff_model_tflite);
  if (model->version() != TFLITE_SCHEMA_VERSION) {
    Serial.printf("# model schema %lu != supported %d\n",
                  (unsigned long)model->version(), TFLITE_SCHEMA_VERSION);
    return;
  }
  // Only the operators this network uses, which keeps the binary small.
  static tflite::MicroMutableOpResolver<4> resolver;
  resolver.AddFullyConnected();
  resolver.AddSoftmax();
  resolver.AddQuantize();
  resolver.AddDequantize();

  static tflite::MicroInterpreter si(model, resolver, arena, kArenaSize);
  interpreter = &si;
  if (interpreter->AllocateTensors() != kTfLiteOk) {
    Serial.println(F("# AllocateTensors failed"));
    return;
  }
  inTensor  = interpreter->input(0);
  outTensor = interpreter->output(0);
  tfliteReady = true;
  Serial.printf("# TFLite Micro ready. arena used %u of %d bytes\n",
                (unsigned)interpreter->arena_used_bytes(), kArenaSize);
}

/* Normalise, quantise, invoke, dequantise. Returns true for leak. */
static bool runModel(const float *feat, float *conf) {
  if (!tfliteReady) { *conf = 0; return false; }

  const float iScale = inTensor->params.scale;
  const int   iZero  = inTensor->params.zero_point;

  for (int i = 0; i < SNIFF_N_FEATURES; i++) {
    float z = (feat[i] - SNIFF_FEAT_MEAN[i]) / SNIFF_FEAT_STD[i];
    int q = (int)lroundf(z / iScale) + iZero;
    if (q < -128) q = -128;
    if (q >  127) q =  127;
    inTensor->data.int8[i] = (int8_t)q;
  }

  if (interpreter->Invoke() != kTfLiteOk) { *conf = 0; return false; }

  const float oScale = outTensor->params.scale;
  const int   oZero  = outTensor->params.zero_point;
  float pSafe = (outTensor->data.int8[0] - oZero) * oScale;
  float pLeak = (outTensor->data.int8[1] - oZero) * oScale;
  *conf = pLeak;
  return pLeak > pSafe;
}

// ------------------------------------------------------------ sensing
static float readAoutVolts() {
  uint32_t acc = 0;
  for (int i = 0; i < ADC_SAMPLES; i++) {
    acc += analogReadMilliVolts(MQ_PIN);
    delayMicroseconds(200);
  }
  return ((acc / (float)ADC_SAMPLES) / 1000.0f) * DIVIDER_RATIO;
}

static float computeRs(float v) {
  if (v <= 0.01f) return NAN;
  if (v >= VC - 0.01f) return 0.0f;
  return RL_KOHM * (VC - v) / v;
}

static float computeSlope(float rs) {
  rsWindow[wIdx] = rs;
  int oldest = (wIdx + 1) % SLOPE_WINDOW;
  wIdx = (wIdx + 1) % SLOPE_WINDOW;
  if (wIdx == 0) wFull = true;
  if (!wFull) return 0.0f;
  return (rs - rsWindow[oldest]) / (float)(SLOPE_WINDOW - 1);
}

static void calibrateRo() {
  Serial.println(F("# calibrating Ro. Clean air, 60 s."));
  float acc = 0; int n = 0;
  for (int i = 0; i < 60; i++) {
    float rs = computeRs(readAoutVolts());
    if (!isnan(rs) && rs > 0) { acc += rs; n++; }
    Serial.printf("# %2d/60  Rs = %.2f kohm\n", i + 1, rs);
    delay(1000);
  }
  if (!n) { Serial.println(F("# FAILED: check the 5 V heater rail.")); return; }
  Ro_kohm = acc / n;
  prefs.begin("sniff", false); prefs.putFloat("ro", Ro_kohm); prefs.end();
  Serial.printf("# Ro = %.2f kohm, saved\n", Ro_kohm);
}

static void handleRoot() { server.send_P(200, "text/html", PAGE); }

static void handleData() {
  char b[280];
  snprintf(b, sizeof(b),
    "{\"rr\":%.4f,\"t\":%.2f,\"h\":%.2f,\"d\":%.5f,"
    "\"ai\":%d,\"conf\":%.4f,\"th\":%d,\"cp\":%d,\"rsd\":%.4f,\"ft\":%.4f,\"ro\":%.2f,\"n\":%lu}",
    isnan(curRsRo) ? 0.0f : curRsRo,
    isnan(lastTemp) ? 0.0f : lastTemp,
    isnan(lastRH) ? 0.0f : lastRH,
    curSlope, curAlarm ? 1 : 0, curConf, curThresh ? 1 : 0, curComp ? 1 : 0,
    curResid, (double)BASELINE_THRESHOLD, Ro_kohm, (unsigned long)sampleCount);
  server.send(200, "application/json", b);
}

// ------------------------------------------------------------ setup
void setup() {
  Serial.begin(115200);
  delay(400);

  pinMode(DHT_PWR, OUTPUT); digitalWrite(DHT_PWR, HIGH); delay(100);
  pinMode(LED_ALARM, OUTPUT); pinMode(LED_SAFE, OUTPUT); pinMode(BUZZER_PIN, OUTPUT);
  digitalWrite(LED_ALARM, LOW); digitalWrite(LED_SAFE, LOW); digitalWrite(BUZZER_PIN, LOW);

  // ADC1 is mandatory here: ADC2 is taken over by the Wi-Fi driver, and this
  // build runs Wi-Fi, so the gas channel has to be on GPIO32-39.
  analogSetPinAttenuation(MQ_PIN, ADC_11db);
  dht.begin();

  prefs.begin("sniff", true); Ro_kohm = prefs.getFloat("ro", 0.0f); prefs.end();
  for (int i = 0; i < SLOPE_WINDOW; i++) rsWindow[i] = 0.0f;

  Serial.println(F("\n# SNIFF - TinyML gas detector with Wi-Fi dashboard"));
  setupModel();
  if (Ro_kohm <= 0) Serial.println(F("# WARNING: Ro not calibrated. Send 'c' in clean air."));
  else Serial.printf("# Ro = %.2f kohm\n", Ro_kohm);

  WiFi.mode(WIFI_AP);
  WiFi.softAP(AP_SSID, AP_PASS);
  Serial.printf("# Wi-Fi AP \"%s\"  password \"%s\"\n", AP_SSID, AP_PASS);
  Serial.print(F("# dashboard: http://")); Serial.println(WiFi.softAPIP());

  server.on("/", handleRoot);
  server.on("/d", handleData);
  server.begin();

  Serial.println(F("# rs_ro,temp_c,rh_pct,drs_dt,model,confidence,fixed_says,compensated_says"));
}

// ------------------------------------------------------------ loop
void loop() {
  server.handleClient();

  if (Serial.available()) {
    char c = Serial.read();
    if (c == 'c') calibrateRo();
    else if (c == 'r') Serial.printf("# Ro = %.2f kohm\n", Ro_kohm);
    else if (c == 'm') { muted = !muted; Serial.printf("# muted=%d\n", muted); }
    else if (c == 't') {
      // Exercises the whole output stage without needing gas.
      testUntil = millis() + 4000;
      digitalWrite(LED_ALARM, HIGH);
      digitalWrite(LED_SAFE, LOW);
      if (!muted) digitalWrite(BUZZER_PIN, HIGH);
      Serial.println(F("# SELF-TEST: alarm asserted for 4 s"));
    }
  }

  // End the self-test promptly rather than on the next 1 Hz tick.
  if (testUntil && millis() >= testUntil) {
    testUntil = 0;
    bool a = curAlarm && (millis() >= WARMUP_MS);
    digitalWrite(LED_ALARM, a ? HIGH : LOW);
    digitalWrite(LED_SAFE,  a ? LOW  : HIGH);
    digitalWrite(BUZZER_PIN, (a && !muted) ? HIGH : LOW);
    Serial.println(F("# SELF-TEST complete, outputs restored"));
  }

  unsigned long now = millis();
  if (now - lastLoop < LOOP_MS) return;
  lastLoop = now;

  if (now - lastDht >= DHT_MIN_MS) {
    float t = dht.readTemperature(), h = dht.readHumidity();
    if (!isnan(t)) lastTemp = t;
    if (!isnan(h)) lastRH = h;
    lastDht = now;
  }

  float rs = computeRs(readAoutVolts());
  if (isnan(rs) || Ro_kohm <= 0 || isnan(lastTemp) || isnan(lastRH)) {
    Serial.println(F("# waiting for valid sensor data"));
    return;
  }

  curRsRo  = rs / Ro_kohm;
  curSlope = computeSlope(rs);

  float feat[SNIFF_N_FEATURES] = { curRsRo, lastTemp, lastRH, curSlope };
  bool leak = runModel(feat, &curConf);

  leakStreak = leak ? (leakStreak + 1) : 0;
  curAlarm   = (leakStreak >= CONFIRM_SAMPLES);
  curThresh  = (curRsRo < BASELINE_THRESHOLD);

  // Compensated detector: compare Rs/Ro against what clean air SHOULD read at
  // this temperature and humidity, rather than against a constant.
  float expected = SNIFF_COMP_A + SNIFF_COMP_B * lastTemp + SNIFF_COMP_C * lastRH;
  curResid = curRsRo - expected;
  curComp  = (curResid < SNIFF_COMP_T);
  sampleCount++;

  bool warming   = (millis() < WARMUP_MS);
  bool assertOut = (curAlarm && !warming) || (testUntil && millis() < testUntil);
  digitalWrite(LED_ALARM, assertOut ? HIGH : LOW);
  digitalWrite(LED_SAFE,  assertOut ? LOW  : HIGH);
  digitalWrite(BUZZER_PIN, (assertOut && !muted) ? HIGH : LOW);

  if (warming)
    Serial.printf("# warming up, alarm suppressed, %lus left\n",
                  (unsigned long)((WARMUP_MS - millis()) / 1000));

  Serial.printf("%.4f,%.2f,%.2f,%.5f,%s,%.3f,%s,%s\n",
                curRsRo, lastTemp, lastRH, curSlope,
                curAlarm ? "LEAK" : "safe", curConf,
                curThresh ? "LEAK" : "safe",
                curComp   ? "LEAK" : "safe");
}
